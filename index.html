grid_optimizer.py 
from voltmind import SmartGrid, AIDispatcher

# AI-balanced load distribution
grid = SmartGrid(region="EU-WEST")
dispatcher = AIDispatcher(model="volt-v3")

while grid.online:
    load = grid.forecast(horizon=24)
    plan = dispatcher.optimize(load)
    grid.dispatch(plan)
    print(f"Δ Efficiency: {plan.gain:.1%}")
